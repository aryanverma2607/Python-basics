MySQL is not case sensitive language. i.e queries can run in both upper and lower case


select version();  # it will shoa the version of MySQL
select now();  #it will show the current date and time
select curtime(); #it will show only current time
select database(); #it selects the database from databases
select @@hostname;   #it return the host as localhost
select @@port;  #it return the port number on which MySQL is running

select 10 + 20;   #it calculate the addition and return 30
select sqrt(49);     #it returns the square root of 49 that is 7
select pow(2,5);  #it return 2 to the power 5 as output that is 32
select rand();  #it generates the random number in decimal form
system cls;  #it clear screen



==========database creation==========
1. create database database-name;
2. CREATE DATABASE IF NOT EXISTS DATABASE-NAME; #IT CREATE DATABASE IF NOT EXIST IF ALREADY EXISTS THEN IT WILL NOT GIVE ANY ERROR
3. DROP DATABASE DATABASE-NAME; #IT WILL DELETE ENTIRE DATABASE COMPLETELY
4. USE DATABASE-NAME;
