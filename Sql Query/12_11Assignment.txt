3. Total Projects
4. Total Project Budget

mysql> SELECT
    ->     d.dept_name,
    ->     COALESCE(e.total_employees, 0) AS total_employees,
    ->     COALESCE(p.total_projects, 0) AS total_projects,
    ->     COALESCE(p.total_budget, 0) AS total_project_budget,
    ->     COALESCE(w.total_hours, 0) AS total_working_hours
    -> FROM department AS d
    ->
    -> LEFT JOIN (
    ->     SELECT dept_id, COUNT(*) AS total_employees
    ->     FROM employee
    ->     GROUP BY dept_id
    -> ) AS e
    -> ON d.dept_id = e.dept_id
    ->
    -> LEFT JOIN (
    ->     SELECT dept_id,
    ->            COUNT(*) AS total_projects,
    ->            SUM(budget) AS total_budget

    -> LEFT JOIN (

    ->     FROM project AS p
    ->     JOIN employee_project AS ep

    -> ON d.dept_id = w.dept_id;
